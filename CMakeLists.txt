# Jason Stys: portable fault-policy kernel and Linux-only owned-simulator relay. Map: docs/SOURCE_MAP.md.
cmake_minimum_required(VERSION 3.25)
project(sitl_assurance VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LAB_SANITIZE "Instrument authored C++ with address/undefined sanitizers" OFF)
add_library(fault_core native/fault.cpp)
target_include_directories(fault_core PUBLIC native)
if(MSVC)
  target_compile_options(fault_core PUBLIC /W4 /WX /permissive-)
else()
  target_compile_options(fault_core PUBLIC -Wall -Wextra -Wpedantic -Werror)
  if(LAB_SANITIZE)
    target_compile_options(fault_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fault_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(core_tests native/core_tests.cpp)
target_link_libraries(core_tests PRIVATE fault_core)
enable_testing()
add_test(NAME bounded_fault_kernel COMMAND core_tests)
set_tests_properties(bounded_fault_kernel PROPERTIES TIMEOUT 30)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(fault_relay native/relay.cpp)
  target_link_libraries(fault_relay PRIVATE fault_core)
endif()
